"use strict"; document.addEventListener("DOMContentLoaded", function () { const form = document.getElementById("chat-form"); const input = document.getElementById("msg"); const chat = document.getElementById("chat"); const status = document.getElementById("status"); function reply(text, sender) { const box = document.createElement("div"); box.className = sender === "user" ? "message user-message" : "message assistant-message"; const label = document.createElement("div"); label.className = "message-label"; label.textContent = sender === "user" ? "BOSS" : "J.A.R.V.I.S"; const p = document.createElement("p"); p.textContent = text; box.append(label, p); chat.appendChild(box); chat.scrollTop = chat.scrollHeight; } if (!form || !input || !chat) { alert("JARVIS ERROR: Check index.html element IDs."); return; } status.textContent = "READY"; form.addEventListener("submit", function (event) { event.preventDefault(); const command = input.value.trim(); if (!command) return; input.value = ""; reply(command, "user"); const cmd = command.toLowerCase(); let answer; if (cmd.includes("time")) { answer = "The time is " + new Date().toLocaleTimeString(); } else if (cmd.includes("date")) { answer = "Today's date is " + new Date().toLocaleDateString(); } else if (cmd.includes("youtube")) { answer = "Opening YouTube, Boss."; window.open("https://www.youtube.com/", "_blank"); } else if (cmd.includes("google")) { answer = "Opening Google, Boss."; window.open("https://www.google.com/", "_blank"); } else if (cmd.includes("instagram")) { answer = "Opening Instagram, Boss."; window.open("https://www.instagram.com/", "_blank"); } else if (cmd.includes("whatsapp")) { answer = "Opening WhatsApp, Boss."; window.open("https://wa.me/", "_blank"); } else if (cmd.includes("hello") || cmd.includes("hi")) { answer = "Hello Boss! JARVIS command system is working."; } else { answer = "Command received, Boss. Basic system is working. AI connection is not enabled in this test."; } reply(answer, "assistant"); }); });
